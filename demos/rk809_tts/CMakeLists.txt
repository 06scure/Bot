cmake_minimum_required(VERSION 3.16)
project(rk809_tts_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(YOLO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../yolo_camera" CACHE PATH "YOLO demo dependencies")
find_package(Threads REQUIRED)
add_library(speech STATIC audio.cpp tts.cpp)
target_include_directories(speech PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE "${YOLO_DIR}/3rdparty/rknpu2/include")
target_link_libraries(speech PUBLIC "${YOLO_DIR}/3rdparty/rknpu2/lib/librknnrt.so" Threads::Threads)
target_compile_options(speech PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rk809_tts_demo main.cpp)
target_link_libraries(rk809_tts_demo PRIVATE speech)
