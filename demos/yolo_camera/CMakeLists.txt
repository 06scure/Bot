cmake_minimum_required(VERSION 3.16)
project(yolo_camera_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

option(CAMERA_CORE_ONLY "Build hardware-independent checks only" OFF)
if(BUILD_TESTING)
    add_executable(camera_core_test core_test.cpp)
    add_test(NAME camera_core COMMAND camera_core_test)
endif()
if(CAMERA_CORE_ONLY)
    return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build the demo on the Linux board, or use CAMERA_CORE_ONLY for checks")
endif()
set(LUBANCAT_EXAMPLES "/root/lubancat_ai_manual_code/example" CACHE PATH "Vendor example directory")
set(RKNN_RUNTIME "${LUBANCAT_EXAMPLES}/3rdparty/rknpu2/Linux/aarch64/librknnrt.so" CACHE FILEPATH "Known working ARM64 Runtime")
set(RKNN_INCLUDE "${LUBANCAT_EXAMPLES}/3rdparty/rknpu2/include" CACHE PATH "RKNN headers")
set(YOLO_DIR "${LUBANCAT_EXAMPLES}/yolo26/cpp")
foreach(required "${RKNN_RUNTIME}" "${RKNN_INCLUDE}/rknn_api.h" "${YOLO_DIR}/postprocess_det.cc")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing dependency: ${required}")
    endif()
endforeach()
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui imgcodecs)
add_library(rknn_runtime SHARED IMPORTED)
set_target_properties(rknn_runtime PROPERTIES IMPORTED_LOCATION "${RKNN_RUNTIME}")
# Compile the vendor postprocessor without changing the user's working vendor tree.
# Its per-frame printf causes excessive SSH traffic; silence this one diagnostic locally.
file(READ "${YOLO_DIR}/postprocess_det.cc" vendor_postprocess)
string(REPLACE "printf(\"validCount=%d\\n\", validCount);" "/* per-frame diagnostic omitted */" vendor_postprocess "${vendor_postprocess}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/postprocess_det.cc" "${vendor_postprocess}")
add_executable(yolo_camera_demo main.cpp camera.cpp detector.cpp "${CMAKE_CURRENT_BINARY_DIR}/postprocess_det.cc")
target_include_directories(yolo_camera_demo PRIVATE "${YOLO_DIR}" "${LUBANCAT_EXAMPLES}/utils" "${RKNN_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yolo_camera_demo PRIVATE rknn_runtime ${OpenCV_LIBS})
target_compile_options(yolo_camera_demo PRIVATE -Wall -Wextra)
set_target_properties(yolo_camera_demo PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
# Configure-time copy ensures a changed Runtime is picked up on the next build.sh.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
configure_file("${RKNN_RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}/lib/librknnrt.so" COPYONLY)
